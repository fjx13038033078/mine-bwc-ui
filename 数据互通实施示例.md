### 1.1 设备台账模块（equipment/ledger/index.vue）

#### 原有代码（数据孤立）
```vue
<script setup>
// ❌ 数据在组件内部维护

  deviceList.value.push(device);
  // 其他模块无法感知这个变化
```

#### 改造后代码（使用Store）
```vue
<script setup>
import { useCopperMineStore } from '@/store/modules/copperMineData';

// ✅ 使用Store管理数据
const store = useCopperMineStore();

// 从Store获取设备列表（响应式）
const deviceList = computed(() => store.devices);

// 添加设备通过Store
const addDevice = (device) => {
  store.addDevice({
    ...device,
    deviceId: generateId('DEV'),
    status: 'online',
    videoCount: 0,
    violationCount: 0,
    configuredSkills: [],
    createTime: formatDateTime(),
    updateTime: formatDateTime()
  });
  
  proxy?.$modal.msgSuccess('设备添加成功');
  // 其他模块（技能配置、视频管理）自动可以看到新设备
};

// 更新设备
const updateDevice = (device) => {
  store.updateDevice(device.deviceId, {
    ...device,
    updateTime: formatDateTime()
  });
  
  proxy?.$modal.msgSuccess('设备更新成功');
};

// 删除设备
const deleteDevice = (deviceId) => {
  proxy?.$modal.confirm('确认删除该设备吗？').then(() => {
    store.deleteDevice(deviceId);
    // Store会自动清理关联数据
    proxy?.$modal.msgSuccess('设备删除成功');
  });
};

// 设备统计（自动计算）
const stats = computed(() => ({
  total: store.devices.length,
  online: store.devices.filter(d => d.status === 'online').length,
  offline: store.devices.filter(d => d.status === 'offline').length,
  maintenance: store.devices.filter(d => d.status === 'maintenance').length,
  faulty: store.devices.filter(d => d.status === 'faulty').length
}));
</script>
```

### 1.3 监控预警模块（violation-monitor/index.vue）

#### 改造后代码（核心！）
```vue
<script setup>
import { useCopperMineStore } from '@/store/modules/copperMineData';

const store = useCopperMineStore();

// ✅ 从Store实时获取违规数据（视频AI识别自动推送）
const violationList = computed(() => {
  let list = store.violations;
  
  // 应用筛选条件
  if (queryParams.value.violationType) {
    list = list.filter(v => v.violationType === queryParams.value.violationType);
  }
  if (queryParams.value.severity) {
    list = list.filter(v => v.severity === queryParams.value.severity);
  }
  if (queryParams.value.status) {
    list = list.filter(v => {
      if (queryParams.value.status === 'pending') return !v.handled;
      if (queryParams.value.status === 'handled') return v.handled;
      return true;
    });
  }
  
  // 分页
  const start = (queryParams.value.pageNum - 1) * queryParams.value.pageSize;
  const end = start + queryParams.value.pageSize;
  return list.slice(start, end);
});

// 违规统计（实时计算）
const stats = computed(() => ({
  total: store.violations.length,
  pending: store.violations.filter(v => !v.handled).length,
  handled: store.violations.filter(v => v.handled).length,
}));

// 处理违规
const handleProcessViolation = (violationId: string) => {
  const violation = store.violations.find(v => v.resultId === violationId);
  if (!violation) return;
  
  // 打开处理对话框
  processForm.value = {
    violationId,
    handler: '',
    handleAction: '',
    handleResult: ''
  };
  processDialog.visible = true;
};

// 提交处理
const submitProcess = () => {
  const { violationId, handler, handleAction, handleResult } = processForm.value;
  
  // 更新违规状态（自动更新设备统计）
  store.updateViolationStatus(violationId, {
    handled: true,
    handler,
    handleTime: formatDateTime(),
    handleAction,
    handleResult
  });
  
  proxy?.$modal.msgSuccess('违规处理完成');
  processDialog.visible = false;
  
  // Store会自动：
  // 1. 更新设备的违规统计
  // 2. 更新综合看板的统计
  // 3. 更新部门违规统计
};

// 查看违规详情（完整信息）
const viewViolationDetail = (violationId: string) => {
  const violation = store.violations.find(v => v.resultId === violationId);
  if (!violation) return;
  
  const video = store.videos.find(v => v.videoId === violation.videoId);
  const device = store.devices.find(d => d.deviceId === video?.deviceId);
  const regulation = store.matchRegulation(violation.violationType);
  const knowledge = store.matchKnowledge(violation.violationType);
  
  // 完整的违规信息（自动关联）
  viewData.value = {
    ...violation,
    videoInfo: video,
    deviceInfo: device,
    regulationInfo: regulation,
    handlingInfo: knowledge
  };
  
  viewDialog.visible = true;
};

// 标记误报
const markFalseAlarm = (violationId: string) => {
  proxy?.$modal.confirm('确认标记为误报吗？').then(() => {
    store.markViolationAsFalseAlarm(violationId);
    // Store会自动减少设备的违规统计
    proxy?.$modal.msgSuccess('已标记为误报');
  });
};
</script>
```

### 1.6 综合看板模块（dashboard/index.vue）

#### 改造后代码
```vue
<script setup>
import { useCopperMineStore } from '@/store/modules/copperMineData';
import * as echarts from 'echarts';

const store = useCopperMineStore();

// ✅ 所有统计数据实时获取
const overallStats = computed(() => ({
  deviceCount: store.devices.length,
  onlineDevices: store.devices.filter(d => d.status === 'online').length,
  totalVideos: store.videos.length,
  totalViolations: store.violations.length,
  pendingViolations: store.violations.filter(v => !v.handled).length,
  enabledSkills: store.aiSkills.filter(s => s.status === 'enabled').length
}));

// 部门违规统计
const departmentStats = computed(() => {
  const stats: Record<string, number> = {};
  
  store.violations.forEach(violation => {
    const video = store.videos.find(v => v.videoId === violation.videoId);
    const device = store.devices.find(d => d.deviceId === video?.deviceId);
    const dept = device?.department || '未知部门';
    
    if (!stats[dept]) stats[dept] = 0;
    stats[dept]++;
  });
  
  return Object.entries(stats).map(([name, value]) => ({ name, value }));
});

// 违规类型TOP10
const violationTypeStats = computed(() => {
  const stats: Record<string, number> = {};
  
  store.violations.forEach(v => {
    if (!stats[v.violationType]) stats[v.violationType] = 0;
    stats[v.violationType]++;
  });
  
  return Object.entries(stats)
    .sort((a, b) => b[1] - a[1])
    .slice(0, 10)
    .map(([name, value]) => ({ name, value }));
});

// 最新违规列表
const latestViolations = computed(() => {
  return store.violations
    .filter(v => !v.handled)
    .sort((a, b) => new Date(b.timestamp).getTime() - new Date(a.timestamp).getTime())
    .slice(0, 10)
    .map(v => {
      const video = store.videos.find(vid => vid.videoId === v.videoId);
      const device = store.devices.find(d => d.deviceId === video?.deviceId);
      return {
        ...v,
        deviceInfo: device,
        captureTime: video?.captureTime
      };
    });
});

// 图表自动更新
watch([departmentStats, violationTypeStats], () => {
  updateCharts();
});

const updateCharts = () => {
  // ECharts图表会自动响应数据变化
  // 无需手动刷新
};
</script>
```

## 三、数据一致性保证

### 3.1 单向数据流

```
User Action (用户操作)
      ↓
Store Action (Store方法)
      ↓
State Update (状态更新)
      ↓
Computed Update (计算属性更新)
      ↓
Component Re-render (组件重新渲染)
```

### 3.2 禁止的操作

```typescript
// ❌ 错误：直接修改Store状态
const store = useCopperMineStore();
store.devices[0].status = 'offline'; // 不应该这样做

// ✅ 正确：通过Store方法修改
store.updateDevice('DEV001', { status: 'offline' });


// ❌ 错误：组件内维护数据副本
const localDevices = ref([...store.devices]); // 会丢失响应性

// ✅ 正确：使用computed
const localDevices = computed(() => store.devices);


// ❌ 错误：手动同步数据
const updateDeviceStatistics = () => {
  // 手动计算统计...
  deviceCount.value = devices.value.length;
};

// ✅ 正确：使用computed自动计算
const deviceCount = computed(() => store.devices.length);
```

## 四、性能优化建议

### 4.1 使用computed缓存

```typescript
// ✅ computed会缓存结果，依赖不变时不重新计算
const expensiveStats = computed(() => {
  return store.violations
    .filter(v => v.severity === 'high')
    .map(v => complexCalculation(v))
    .sort((a, b) => b.score - a.score);
});

// ❌ 每次访问都重新计算
const expensiveStats = () => {
  return store.violations
    .filter(v => v.severity === 'high')
    .map(v => complexCalculation(v))
    .sort((a, b) => b.score - a.score);
};
```

```typescript
// ✅ 只加载需要的数据
```

